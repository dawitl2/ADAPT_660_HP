cmake_minimum_required(VERSION 3.16)
project(adapt_firmware LANGUAGES C CXX)
option(ADAPT_PHYSICAL_TARGET "Build a physical ADAPT 660 image" OFF)
if(ADAPT_PHYSICAL_TARGET)
  message(FATAL_ERROR "SoC/SDK/board and stock restoration are UNKNOWN; physical target disabled")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(MSVC)
  add_compile_options(/W4 /WX)
else()
  add_compile_options(-Wall -Wextra -Wpedantic -Werror)
endif()
enable_testing()
add_subdirectory(../protocol protocol)
add_library(adapt_core src/core/button.cpp src/core/core.cpp src/core/config.cpp src/core/lifecycle.cpp src/core/tones.cpp src/core/gatt.cpp)
target_include_directories(adapt_core PUBLIC include)
target_link_libraries(adapt_core PUBLIC adapt_protocol)
add_executable(button_tests tests/button_tests.cpp)
target_link_libraries(button_tests PRIVATE adapt_core)
add_test(NAME button COMMAND button_tests)
add_executable(core_tests tests/core_tests.cpp)
target_link_libraries(core_tests PRIVATE adapt_core)
add_test(NAME core COMMAND core_tests)
add_executable(config_tests tests/config_tests.cpp)
target_link_libraries(config_tests PRIVATE adapt_core)
add_test(NAME config COMMAND config_tests)
add_executable(lifecycle_tests tests/lifecycle_tests.cpp)
target_link_libraries(lifecycle_tests PRIVATE adapt_core)
add_test(NAME lifecycle COMMAND lifecycle_tests)
add_executable(diagnostic_tests tests/diagnostic_tests.cpp)
target_link_libraries(diagnostic_tests PRIVATE adapt_core)
add_test(NAME diagnostics COMMAND diagnostic_tests)
add_executable(integration_tests tests/integration_tests.cpp)
target_link_libraries(integration_tests PRIVATE adapt_core)
add_test(NAME integration COMMAND integration_tests)
add_executable(tone_tests tests/tone_tests.cpp)
target_link_libraries(tone_tests PRIVATE adapt_core)
add_test(NAME tones COMMAND tone_tests)
add_executable(gatt_tests tests/gatt_tests.cpp)
target_link_libraries(gatt_tests PRIVATE adapt_core)
add_test(NAME gatt COMMAND gatt_tests)
add_library(adapt_sim_platform src/platform/sim/file_settings.cpp)
target_link_libraries(adapt_sim_platform PUBLIC adapt_core)
add_executable(file_settings_tests tests/file_settings_tests.cpp)
target_link_libraries(file_settings_tests PRIVATE adapt_sim_platform)
add_test(NAME file_settings COMMAND file_settings_tests)
add_executable(adapt_sim simulator/main.cpp)
target_link_libraries(adapt_sim PRIVATE adapt_sim_platform)
add_executable(target_adapt660_pending src/platform/target_adapt660_pending/main.cpp)
target_link_libraries(target_adapt660_pending PRIVATE adapt_core)
add_test(NAME pending_target COMMAND target_adapt660_pending)
find_package(Python3 COMPONENTS Interpreter QUIET)
if(Python3_FOUND)
  add_test(NAME simulator COMMAND ${Python3_EXECUTABLE}
    ${CMAKE_CURRENT_SOURCE_DIR}/simulator/test_simulator.py $<TARGET_FILE:adapt_sim>)
  add_test(NAME simulator_phase2 COMMAND ${Python3_EXECUTABLE}
    ${CMAKE_CURRENT_SOURCE_DIR}/simulator/test_phase2.py $<TARGET_FILE:adapt_sim>)
endif()
